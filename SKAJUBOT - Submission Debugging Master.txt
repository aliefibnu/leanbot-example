#include <Leanbot.h>
#include <Arduino_APDS9960.h>
#include <U8g2lib.h>
#define bunyikanBuzzer(jml) while (jml--) Leanbot.tone(1500, 100), delay(150)
U8G2_SH1106_128X64_NONAME_1_HW_I2C u8g2(U8G2_R0);

void OLEDprint(const char *msg) {
  u8g2.firstPage();
  do {
    u8g2.setFont(u8g2_font_7x13B_tr);
    u8g2.drawStr(10, 18, "SKAJU BOT");  u8g2.drawLine(0, 24, 128, 24);
    u8g2.drawStr(10, 45, "Color :");    u8g2.drawStr(10, 62, msg);
  } while (u8g2.nextPage());
}

void setup() {
  Leanbot.begin(); Serial.begin(115200); u8g2.begin();
  while (!APDS.begin()) OLEDprint("Sensor Error");
  OLEDprint("Scanning...");
}

void loop() {
  if (!APDS.colorAvailable()) return OLEDprint("Scanning..."), delay(50);
  int r, g, b, c; APDS.readColor(r, g, b, c);
  if (c < 15) return OLEDprint("Scanning..."), delay(50);

  char buf[40]; sprintf(buf, "R:%d G:%d B:%d C:%d", r, g, b, c); Serial.println(buf);

  const char *warna = "UNKNOWN"; int ketukan = 0;
  if (r > g && r > b)       { warna = "RED";    ketukan = 2; }
  else if (g > r && g > b)  { warna = "GREEN";  ketukan = 5; }
  else if (b > r && b > g)  { warna = "BLUE";   ketukan = 1; }
  else if (r > b && g > b)  { warna = "YELLOW"; ketukan = 3; }

  OLEDprint(warna); bunyikanBuzzer(ketukan); 
  Serial.print("Detected : "); Serial.println(warna);
  delay(500);
}
