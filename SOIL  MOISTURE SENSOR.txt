//SOIL SENSOR
#include <Leanbot.h>
#include <U8g2lib.h>

// Inisialisasi OLED SH1106
U8G2_SH1106_128X64_NONAME_1_HW_I2C u8g2(U8G2_R0);

const int soilPin = A0; 

// Nilai standar (Default), akan terupdate otomatis jika kalibrasi dilakukan
int airValue = 650;   // Semakin besar angka ini, semakin dianggap kering
int waterValue = 300; // Semakin kecil angka ini, semakin dianggap basah

void setup() {
  Leanbot.begin();
  u8g2.begin();
  Serial.begin(115200);

  // Tampilan awal (Splash Screen)
  u8g2.firstPage();
  do {
    u8g2.setFont(u8g2_font_ncenB10_tr);
    u8g2.setCursor(20, 35);
    u8g2.print(F("SIAPKAN SENSOR..."));
  } while (u8g2.nextPage());
  delay(2000);
}

void loop() {
  // 1. Ambil rata-rata pembacaan (supaya angka tidak goyang)
  long sum = 0;
  for(int i=0; i<10; i++){
    sum += analogRead(soilPin);
    delay(10);
  }
  int rawValue = sum / 10;

  // 2. Kirim data ke Serial untuk monitor
  Serial.print("Raw: "); Serial.println(rawValue);

  // 3. Hitung Persentase
  // map(nilai, batas_kering, batas_basah, 0 persen, 100 persen)
  int percentage = map(rawValue, airValue, waterValue, 0, 100);
  percentage = constrain(percentage, 0, 100);

  // 4. Update Tampilan OLED
  u8g2.firstPage();
  do {
    // Header
    u8g2.setFont(u8g2_font_6x12_tr);
    u8g2.setCursor(0, 10);
    u8g2.print(F("MONITOR TANAH (A0)"));

    // Angka Besar
    u8g2.setFont(u8g2_font_logisoso24_tn);
    u8g2.setCursor(25, 45);
    u8g2.print(percentage);
    u8g2.setFont(u8g2_font_ncenB12_tr);
    u8g2.print(F(" %"));

    // Pesan Status & Lampu
    u8g2.setFont(u8g2_font_6x12_tr);
    u8g2.setCursor(0, 62);

    if (percentage < 20) {
      u8g2.print(F("STATUS: KERING !"));
      LbRGB.fillColor(0xFF0000); // Merah
      Leanbot.tone(1000, 50);    // Bunyi bip pendek tanda bahaya
    } 
    else if (percentage >= 20 && percentage <= 70) {
      u8g2.print(F("STATUS: LEMBAP (IDEAL)"));
      LbRGB.fillColor(0x00FF00); // Hijau
    } 
    else {
      u8g2.print(F("STATUS: SANGAT BASAH"));
      LbRGB.fillColor(0x0000FF); // Biru
    }
    LbRGB.show();

  } while (u8g2.nextPage());

  delay(500); 
}
