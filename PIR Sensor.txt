#include <Leanbot.h>
#include <U8g2lib.h>

// Inisialisasi OLED SH1106
U8G2_SH1106_128X64_NONAME_1_HW_I2C u8g2(U8G2_R0);

const int inputPin = A1;      // Pin sensor PIR
int pirState = LOW;

void setup() {
  Leanbot.begin();            // Inisialisasi Leanbot
  pinMode(inputPin, INPUT);   // PIR sebagai input
  Serial.begin(115200);       // Baud rate sesuai file PIR

  u8g2.setBusClock(400000);   // Kecepatan I2C
  u8g2.begin();               // Start OLED
}

void loop() {
  int val = digitalRead(inputPin); // Baca sensor PIR

  if (val == HIGH) {
    // Tampilan OLED saat ada gerakan
    u8g2.firstPage();
    do {
      u8g2.setFont(u8g2_font_ncenB10_tr);
      u8g2.setCursor(15, 30);
      u8g2.print(F("ADA ORANG!")); 
    } while (u8g2.nextPage());

    // Alarm: LED Berkedip & Buzzer Berbunyi
    for (int i = 0; i < 3; i++) {
      LbRGB.fillColor(0xFF0000); // Merah
      LbRGB.show();
      Leanbot.tone(1500, 150);   // Buzzer On
      delay(150);

      LbRGB.fillColor(0xFFFF00); // Kuning
      LbRGB.show();
      Leanbot.tone(800, 150);    // Buzzer Nada Berbeda
      delay(150);
    }
    pirState = HIGH;
  } 
  else {
    // Jika tidak ada gerakan
    if (pirState == HIGH) {
      u8g2.firstPage();
      do {
        u8g2.setFont(u8g2_font_ncenB10_tr);
        u8g2.setCursor(15, 30);
        u8g2.print(F("AMAN")); 
      } while (u8g2.nextPage());
      
      LbRGB.clear();
      LbRGB.show();
      Leanbot.noTone(); // Matikan suara hanya saat transisi dari HIGH ke LOW
      pirState = LOW;
    }
  }
}
